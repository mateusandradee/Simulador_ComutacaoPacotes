    <script>
        const canvas = document.getElementById('simCanvas');
        const ctx = canvas.getContext('2d');

        // Constantes da Rede
        const D_PROC = 15;
        const D_TRANS = 30;
        const D_PROP = 20;
        const ROUTER_SERVICE_INTERVAL = 380;

        // Mapeamento dos Nós da Rede
        const nodes = {
            A:  { id: 'A',  x: 70,  y: 120, label: 'Host A', color: '#10b981', type: 'host' },
            B:  { id: 'B',  x: 70,  y: 300, label: 'Host B', color: '#8b5cf6', type: 'host' },
            
            R1: { id: 'R1', x: 230, y: 210, label: 'Roteador R1', color: '#475569', type: 'router', queue: [], maxBuffer: 6 },
            
            R2: { id: 'R2', x: 475, y: 80,  label: 'Roteador R2', color: '#475569', type: 'router', queue: [], maxBuffer: 6 },
            R3: { id: 'R3', x: 475, y: 210, label: 'Roteador R3', color: '#475569', type: 'router', queue: [], maxBuffer: 6 },
            R4: { id: 'R4', x: 475, y: 340, label: 'Roteador R4', color: '#475569', type: 'router', queue: [], maxBuffer: 6 },
            
            R5: { id: 'R5', x: 720, y: 210, label: 'Roteador R5', color: '#475569', type: 'router', queue: [], maxBuffer: 6 },
            
            C:  { id: 'C',  x: 880, y: 120, label: 'Host C', color: '#0284c7', type: 'host' },
            D:  { id: 'D',  x: 880, y: 300, label: 'Host D', color: '#ec4899', type: 'host' }
        };

        const links = [
            ['A', 'R1'], ['B', 'R1'],
            ['R1', 'R2'], ['R1', 'R3'], ['R1', 'R4'],
            ['R2', 'R5'], ['R3', 'R5'], ['R4', 'R5'],
            ['R5', 'C'], ['R5', 'D']
        ];

        let packets = [];
        let messageBuffers = {};
        let roundRobinIndex = 0;

        function logEvent(msg) {
            const logBox = document.getElementById('logBox');
            const time = new Date().toLocaleTimeString();
            logBox.innerHTML += `<div>[${time}] ${msg}</div>`;
            logBox.scrollTop = logBox.scrollHeight;
        }

        // Função para desenhar setas dos ícones do Roteador
        function drawArrow(ctx, fromx, fromy, tox, toy) {
            const headlen = 4;
            const dx = tox - fromx;
            const dy = toy - fromy;
            const angle = Math.atan2(dy, dx);
            ctx.beginPath();
            ctx.moveTo(fromx, fromy);
            ctx.lineTo(tox, toy);
            ctx.stroke();
            ctx.beginPath();
            ctx.moveTo(tox, toy);
            ctx.lineTo(tox - headlen * Math.cos(angle - Math.PI / 6), toy - headlen * Math.sin(angle - Math.PI / 6));
            ctx.lineTo(tox - headlen * Math.cos(angle + Math.PI / 6), toy - headlen * Math.sin(angle + Math.PI / 6));
            ctx.fill();
        }

        // Renderização dos Nós (Ícones de Computador e Roteador)
        function drawNetwork() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // 1. Links
            ctx.strokeStyle = '#cbd5e1';
            ctx.lineWidth = 2;
            links.forEach(([src, dst]) => {
                ctx.beginPath();
                ctx.moveTo(nodes[src].x, nodes[src].y);
                ctx.lineTo(nodes[dst].x, nodes[dst].y);
                ctx.stroke();
            });

            // 2. Nós
            Object.values(nodes).forEach(node => {
                if (node.type === 'host') {
                    drawHostIcon(node);
                } else if (node.type === 'router') {
                    drawRouterIcon(node);
                    drawRouterBuffer(node);
                }
            });

            // 3. Pacotes em Movimento
            packets.forEach(p => p.draw());
        }

        // Ícone de Host (Usuário / Computador)
        function drawHostIcon(node) {
            // Círculo de fundo colorido identificador
            ctx.beginPath();
            ctx.arc(node.x, node.y, 22, 0, Math.PI * 2);
            ctx.fillStyle = node.color;
            ctx.fill();

            // Desenhar Monitor / Computador em Branco
            ctx.strokeStyle = '#ffffff';
            ctx.fillStyle = '#ffffff';
            ctx.lineWidth = 1.8;

            // Tela do Monitor
            ctx.strokeRect(node.x - 9, node.y - 10, 18, 12);
            // Suporte
            ctx.beginPath();
            ctx.moveTo(node.x - 3, node.y + 2);
            ctx.lineTo(node.x - 3, node.y + 6);
            ctx.moveTo(node.x + 3, node.y + 2);
            ctx.lineTo(node.x + 3, node.y + 6);
            ctx.stroke();
            // Base
            ctx.beginPath();
            ctx.moveTo(node.x - 8, node.y + 6);
            ctx.lineTo(node.x + 8, node.y + 6);
            ctx.stroke();

            // Rótulo
            ctx.fillStyle = '#0f172a';
            ctx.font = '600 12px sans-serif';
            ctx.textAlign = 'center';
            ctx.fillText(node.label, node.x, node.y + 38);
        }

        // Ícone de Roteador (Símbolo Padrão de Redes com Setas Cruzadas)
        function drawRouterIcon(node) {
            // Círculo base do roteador
            ctx.beginPath();
            ctx.arc(node.x, node.y, 23, 0, Math.PI * 2);
            ctx.fillStyle = '#475569';
            ctx.fill();
            ctx.lineWidth = 1.5;
            ctx.strokeStyle = '#334155';
            ctx.stroke();

            // Setas Cruzadas
            ctx.strokeStyle = '#ffffff';
            ctx.fillStyle = '#ffffff';
            ctx.lineWidth = 1.5;

            // Setas Entrando e Saindo
            drawArrow(ctx, node.x - 14, node.y - 5, node.x - 3, node.y - 2);
            drawArrow(ctx, node.x + 14, node.y + 5, node.x + 3, node.y + 2);
            drawArrow(ctx, node.x - 3, node.y + 2, node.x - 14, node.y + 5);
            drawArrow(ctx, node.x + 3, node.y - 2, node.x + 14, node.y - 5);

            // Rótulo
            ctx.fillStyle = '#334155';
            ctx.font = '600 11px sans-serif';
            ctx.textAlign = 'center';
            ctx.fillText(node.label, node.x, node.y + 38);
        }

        // Buffer FIFO no Roteador
        function drawRouterBuffer(router) {
            const bx = router.x - 35;
            const by = router.y - 60;
            const bw = 70;
            const bh = 18;

            ctx.fillStyle = '#ffffff';
            ctx.fillRect(bx, by, bw, bh);
            ctx.strokeStyle = router.queue.length > 3 ? '#ef4444' : '#cbd5e1';
            ctx.lineWidth = 1;
            ctx.strokeRect(bx, by, bw, bh);

            // Indicadores IN/OUT
            ctx.fillStyle = '#16a34a';
            ctx.font = 'bold 8px sans-serif';
            ctx.fillText('IN►', bx - 10, by + 12);

            ctx.fillStyle = '#0284c7';
            ctx.fillText('►OUT', bx + bw + 12, by + 12);

            // Slots do Buffer
            const slots = router.maxBuffer;
            const slotW = bw / slots;
            for (let i = 0; i < slots; i++) {
                ctx.strokeStyle = '#f1f5f9';
                ctx.strokeRect(bx + (i * slotW), by, slotW, bh);

                if (i < router.queue.length) {
                    let pkt = router.queue[i];
                    ctx.fillStyle = pkt.color;
                    ctx.fillRect(bx + (i * slotW) + 2, by + 2, slotW - 4, bh - 4);
                }
            }

            // Exibição de Atraso de Fila
            let currentDQueue = router.queue.length * D_TRANS;
            ctx.fillStyle = currentDQueue > 0 ? '#dc2626' : '#64748b';
            ctx.font = 'bold 9px sans-serif';
            ctx.fillText(`Fila: ${router.queue.length}/${slots} (${currentDQueue}ms)`, router.x, by - 5);
        }

        class Packet {
            constructor(msgId, seqId, totalSeq, data, srcId, dstId, isAck = false, origPath = null) {
                this.msgId = msgId;
                this.seqId = seqId;
                this.totalSeq = totalSeq;
                this.data = data;
                this.srcId = srcId;
                this.dstId = dstId;
                this.isAck = isAck;
                this.startTime = performance.now();
                this.color = isAck ? '#16a34a' : nodes[srcId].color;

                if (!isAck) {
                    const coreRouters = ['R2', 'R3', 'R4'];
                    const chosenCore = coreRouters[roundRobinIndex % 3];
                    roundRobinIndex++;
                    this.path = [srcId, 'R1', chosenCore, 'R5', dstId];
                } else {
                    this.path = origPath ? [...origPath].reverse() : [dstId, 'R5', 'R3', 'R1', srcId];
                }

                this.pathStep = 0;
                this.x = nodes[this.path[0]].x;
                this.y = nodes[this.path[0]].y;
                this.targetX = nodes[this.path[1]].x;
                this.targetY = nodes[this.path[1]].y;
                this.speed = 1.5; //velocidade do pacote ******
                this.state = 'moving';
            }

            update() {
                if (this.state === 'moving') {
                    let dx = this.targetX - this.x;
                    let dy = this.targetY - this.y;
                    let dist = Math.hypot(dx, dy);

                    if (dist < this.speed) {
                        this.x = this.targetX;
                        this.y = this.targetY;
                        this.pathStep++;

                        let currentNodeId = this.path[this.pathStep];
                        let currentNode = nodes[currentNodeId];

                        if (currentNode.type === 'router') {
                            this.state = 'queued';
                            currentNode.queue.push(this);
                        } else if (currentNodeId === this.dstId) {
                            this.state = 'finished';
                            handlePacketArrival(this);
                        }
                    } else {
                        this.x += (dx / dist) * this.speed;
                        this.y += (dy / dist) * this.speed;
                    }
                }
            }

            draw() {
                if (this.state === 'finished' || this.state === 'queued') return;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.isAck ? 6 : 8, 0, Math.PI * 2);
                ctx.fillStyle = this.color;
                ctx.fill();
                ctx.lineWidth = 1;
                ctx.strokeStyle = '#ffffff';
                ctx.stroke();

                ctx.fillStyle = '#ffffff';
                ctx.font = 'bold 8px sans-serif';
                ctx.textAlign = 'center';
                ctx.fillText(this.isAck ? 'ACK' : `P${this.seqId}`, this.x, this.y + 3);
            }
        }

        // Processamento das filas nos Roteadores
        Object.values(nodes).filter(n => n.type === 'router').forEach(router => {
            setInterval(() => {
                if (router.queue.length > 0) {
                    let p = router.queue.shift();
                    let nextNodeId = p.path[p.pathStep + 1];
                    p.targetX = nodes[nextNodeId].x;
                    p.targetY = nodes[nextNodeId].y;
                    p.state = 'moving';
                }
            }, ROUTER_SERVICE_INTERVAL);
        });

        function handlePacketArrival(p) {
            const now = performance.now();

            if (!p.isAck) {
                if (!messageBuffers[p.msgId]) {
                    messageBuffers[p.msgId] = {
                        received: 0,
                        total: p.totalSeq,
                        startTime: p.startTime,
                        srcId: p.srcId,
                        dstId: p.dstId,
                        path: p.path
                    };
                }

                messageBuffers[p.msgId].received++;

                if (messageBuffers[p.msgId].received === messageBuffers[p.msgId].total) {
                    const realTime = (now - messageBuffers[p.msgId].startTime).toFixed(0);
                    document.getElementById('realTimeText').innerText = `${realTime} ms`;
                    logEvent(`Mensagem [${p.msgId}] recebida em ${nodes[p.dstId].label}! Tempo Real: ${realTime}ms.`);

                    logEvent(`Gerando pacote ACK (${p.dstId} → ${p.srcId})...`);
                    let ackPacket = new Packet(p.msgId, 0, 1, 'ACK', p.dstId, p.srcId, true, messageBuffers[p.msgId].path);
                    ackPacket.startTime = messageBuffers[p.msgId].startTime;
                    packets.push(ackPacket);
                }
            } else {
                const rtt = (now - p.startTime).toFixed(0);
                document.getElementById('rttText').innerText = `${rtt} ms`;
                logEvent(`<strong>ACK Confirmado em ${nodes[p.dstId].label}!</strong> RTT: ${rtt}ms.`);
            }
        }

        function calculateEstimatedTime(numPackets, hops) {
            const totalProc = hops * D_PROC;
            const totalProp = hops * D_PROP;
            const totalTrans = numPackets * D_TRANS * 2.2;
            return Math.round(totalProc + totalProp + totalTrans);
        }

        function dispatchMessage(srcId, dstId, text, msgId) {
            const chunkSize = 4;
            const chunks = [];
            for (let i = 0; i < text.length; i += chunkSize) {
                chunks.push(text.substring(i, i + chunkSize));
            }

            const hops = 4;
            const estimatedTime = calculateEstimatedTime(chunks.length, hops);
            document.getElementById('estTimeText').innerText = `~${estimatedTime} ms`;

            logEvent(`Transmissão iniciada: ${nodes[srcId].label} → ${nodes[dstId].label} (${chunks.length} pacotes).`);

            chunks.forEach((chunk, index) => {
                setTimeout(() => {
                    let p = new Packet(msgId, index + 1, chunks.length, chunk, srcId, dstId);
                    packets.push(p);
                }, index * 220);
            });
        }

        function sendCustomMessage() {
            const srcId = document.getElementById('srcHost').value;
            const dstId = document.getElementById('dstHost').value;
            const text = document.getElementById('msgText').value;

            if (!text) return;
            const msgId = 'MSG_' + Math.floor(Math.random() * 1000);
            dispatchMessage(srcId, dstId, text, msgId);
        }

        function loadScenario(type) {
            packets = [];
            Object.values(nodes).forEach(n => { if (n.queue) n.queue = []; });
            
            if (type === 'single') {
                logEvent("--- Cenário 1: Mensagem Única ---");
                dispatchMessage('A', 'C', 'Envio de dados simples', 'MSG_CEN1');
            } else if (type === 'congestion') {
                logEvent("--- Cenário 2: Congestionamento Simultâneo (A→C e B→D) ---");
                dispatchMessage('A', 'C', 'Carga de Dados do Host A para o C', 'MSG_CONG_A');
                dispatchMessage('B', 'D', 'Carga de Dados do Host B para o D', 'MSG_CONG_B');
            }
        }

        function loop() {
            packets.forEach(p => p.update());
            drawNetwork();

            packets = packets.filter(p => p.state !== 'finished');
            document.getElementById('activePacketsText').innerText = packets.length;

            let totalQueueLength = Object.values(nodes)
                .filter(n => n.type === 'router')
                .reduce((acc, r) => acc + r.queue.length, 0);
            
            let currentQueueDelay = totalQueueLength * D_TRANS;
            document.getElementById('queueDelayText').innerText = `${currentQueueDelay} ms`;

            requestAnimationFrame(loop);
        }

        loop();
    </script>
